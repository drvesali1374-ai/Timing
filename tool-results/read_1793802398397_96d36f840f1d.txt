   110→/** تجزیه امن JSON شیء‌شکل */
   111→function parseJsonObject(text: string | null): Record<string, unknown> | null {
   112→  if (!text) return null
   113→  try {
   114→    const parsed: unknown = JSON.parse(text)
   115→    return typeof parsed === 'object' && parsed !== null ? (parsed as Record<string, unknown>) : null
   116→  } catch {
   117→    // JSON نامعتبر — مقدار نا معتبر نادیده گرفته می‌شود
   118→    return null
   119→  }
   120→}

   376→function CalcSettingsCard({ projectId }: { projectId: string }) {
   377→  const qc = useQueryClient()
   378→  const { data, isLoading, error } = useQuery({
   379→    queryKey: qk.projectCalcSettings(projectId),
   380→    queryFn: () => projectsApi.calcSettings(projectId),
   381→  })
   382→
   383→  const [calc, setCalc] = useState<{ allowanceMethod: string; aggregationMethod: string } | null>(
   384→    null
   385→  )
   386→  useEffect(() => {
   387→    if (data) setCalc({ allowanceMethod: data.allowanceMethod, aggregationMethod: data.aggregationMethod })
   388→  }, [data])
   389→
   390→  const applyCalc = async (patch: { allowanceMethod?: string; aggregationMethod?: string }) => {
   391→    if (!calc) return
   392→    setCalc({ ...calc, ...patch })
   393→    try {
   394→      await projectsApi.putCalcSettings(projectId, patch)
   395→      void qc.invalidateQueries({ queryKey: qk.projectCalcSettings(projectId) })
   396→      void qc.invalidateQueries({ queryKey: qk.projectSummary(projectId) })
   397→      toast.success('تنظیمات محاسبات ذخیره شد')
   398→    } catch (err) {
   399→      toast.error(err instanceof Error ? err.message : 'ذخیره تنظیمات محاسبات ناموفق بود')
   400→      // بازگشت به مقدار سرور

   402→    }
   403→  }

   406→    <section className="card-premium p-5" aria-label="تنظیمات محاسبات">
   407→      <div className="flex items-center gap-3">
   408→        <div className="chip-icon bg-cyan-soft text-cyan">

   414→        </div>
   415→      </div>
   416→
   417→      {isLoading ? (
   418→        <div className="mt-4 space-y-3">
   419→          <Skeleton className="h-20 w-full rounded-xl" />
   420→          <Skeleton className="h-20 w-full rounded-xl" />
   421→          <Skeleton className="h-10 w-full rounded-xl" />
   422→        </div>
   423→      ) : error ? (
   424→        <div className="mt-4 flex flex-wrap items-center justify-between gap-2 rounded-xl bg-red-soft/60 p-3 text-xs text-red">
   425→          <span>{error instanceof Error ? error.message : 'خطا در بارگذاری تنظیمات محاسبات'}</span>
   426→          <Button
   427→            size="sm"
   428→            variant="outline"
   429→            onClick={() => void qc.invalidateQueries({ queryKey: qk.projectCalcSettings(projectId) })}
   430→          >
   431→            تلاش مجدد
   432→          </Button>
   433→        </div>

   436→          {/* روش اعمال مجازها — FR-15 */}

   438→            <div className="flex items-center gap-1">
   439→              <Label>روش اعمال مجازها (A)</Label>
   440→              <HelpDot
   441→                label="روش اعمال مجازها"
   442→                content={{

   444→                  input: 'یکی از دو روش مجاز را انتخاب کنید؛ فرمول هر روش زیر آن نمایش داده می‌شود.',

   446→                }}

   450→              dir="rtl"

   452→              onValueChange={(v) => void applyCalc({ allowanceMethod: v })}
   453→              className="gap-2"
   454→            >
   455→              {ALLOWANCE_METHODS.map((m) => (

   520→function ProjectOperationsCard({
   521→  projectId,
   522→  projectName,
   523→  dependencyCounts,
   524→  onDeleted,
   525→}: {
   526→  projectId: string
   527→  projectName: string
   528→  dependencyCounts: ProjectDetailDto['dependencyCounts']
   529→  onDeleted: () => void
   530→}) {
   531→  const qc = useQueryClient()
   532→  const navigate = useAppStore((s) => s.navigate)
   533→  const [dupOpen, setDupOpen] = useState(false)
   534→  const [deleteOpen, setDeleteOpen] = useState(false)
   535→
   536→  const deps: Record<string, number> = {
   537→    'ایستگاه‌ها': dependencyCounts.stations,
   538→    'فعالیت‌ها': dependencyCounts.activities,
   539→    'عناصر کاری': dependencyCounts.elements,
   540→    'اپراتورها': dependencyCounts.operators,
   541→    'گروه‌های مجاز': dependencyCounts.groups,
   542→    'جلسات زمان‌سنجی': dependencyCounts.sessions,
   543→    'مشاهدات خام': dependencyCounts.observations,
   544→  }
   545→
   546→  const deleteMutation = useMutation({
   547→    mutationFn: () => projectsApi.remove(projectId),
   548→    onSuccess: () => {
   549→      void qc.invalidateQueries({ queryKey: ['projects'] })
   550→      void qc.removeQueries({ queryKey: qk.project(projectId) })
   551→      toast.success('پروژه به‌طور دائمی حذف شد')
   552→      setDeleteOpen(false)
   553→      onDeleted()
   554→    },
   555→    onError: (err) => toast.error(err instanceof Error ? err.message : 'حذف پروژه ناموفق بود'),
   556→  })
   557→
   558→  const exportPackage = () => {

   560→    if (!win) {
   561→      toast.error('مرورگر پنجره دانلود را مسدود کرد؛ مجوز پاپ‌آپ مرورگر را بررسی کنید.')
   562→      return
   563→    }
   564→    toast.success('خروجی بسته آغاز شد', {
   565→      description: `فایل بسته کامل «${projectName}» دانلود می‌شود.`,
   566→    })
   567→  }
   568→
   569→  const onDuplicated = (newProjectId: string, newName: string) => {
   570→    void qc.invalidateQueries({ queryKey: ['projects'] })
   571→    setDupOpen(false)
   572→    toast.success(`پروژه «${newName}» ساخته شد`, {
   573→      description: 'کپی با بخش‌های انتخاب‌شده ایجاد شد.',
   574→      action: { label: 'بازکردن', onClick: () => navigate({ name: 'project', projectId: newProjectId, stage: 'info' }) },
   575→    })
   576→  }
   577→
   578→  return (
   579→    <section className="card-premium border-red/40 p-5" aria-label="عملیات پروژه">
   580→      <div className="flex items-center gap-3">
   581→        <div className="chip-icon bg-red-soft text-red">
   582→          <Database className="size-5" aria-hidden />
   583→        </div>
   584→        <div>
   585→          <h2 className="font-extrabold">عملیات پروژه</h2>
   586→          <p className="text-xs text-muted-foreground">کپی، خروجی بسته کامل و حذف دائمی</p>
   587→        </div>
   588→      </div>
   589→
   590→      <div className="mt-4 divide-y overflow-hidden rounded-xl border border-red/25">
   591→        {/* کپی پروژه */}
   592→        <div className="flex flex-wrap items-center justify-between gap-3 p-3.5">
   593→          <div>
   594→            <p className="text-sm font-bold">کپی پروژه</p>
   595→            <p className="text-xs text-muted-foreground">
   596→              ساخت پروژه جدید از بخش‌های انتخابی این مطالعه؛ پیش‌فرض فقط ساختار و تنظیمات.
   597→            </p>
   598→          </div>
   599→          <Button variant="outline" className="gap-1.5" onClick={() => setDupOpen(true)}>
   600→            <Copy className="size-4" aria-hidden />
   601→            کپی پروژه
   602→          </Button>
   603→        </div>
   604→
   605→        {/* خروجی بسته کامل */}
   606→        <div className="flex flex-wrap items-center justify-between gap-3 p-3.5">
   607→          <div>
   608→            <p className="text-sm font-bold">خروجی بسته کامل</p>

   610→              فایل JSON نسخه‌دار شامل کل داده پروژه با manifest و checksum.

   615→            دانلود بسته
   616→          </Button>
   617→        </div>
   618→

   662→// دیالوگ کپی پروژه با انتخاب بخش‌ها — FR-01
   663→// ═══════════════════════════════════════════════════════════
   664→function DuplicateProjectDialog({
   665→  project,
   666→  onClose,
   667→  onDone,
   668→}: {
   669→  project: { id: string; name: string }
   670→  onClose: () => void
   671→  onDone: (newProjectId: string, newName: string) => void
   672→}) {
   673→  const [newName, setNewName] = useState(`${project.name} (کپی)`)
   674→  const [sections, setSections] = useState<DuplicateSections>({
   675→    structureAndSettings: true,
   676→    operators: false,
   677→    groupsAndAllowances: false,
   678→    ratingTables: false,
   679→    rawObservations: false,
   680→  })
   681→  const [saving, setSaving] = useState(false)
   682→
   683→  const rows: { key: keyof DuplicateSections; label: string; desc: string }[] = [
   684→    {
   685→      key: 'structureAndSettings',
   686→      label: 'ساختار و تنظیمات',
   687→      desc: 'ایستگاه‌ها، فعالیت‌ها، عناصر کاری و تنظیمات محاسبات',
   688→    },
   689→    { key: 'operators', label: 'اپراتورها', desc: 'فهرست اپراتورهای این مطالعه' },
   690→    {
   691→      key: 'groupsAndAllowances',
   692→      label: 'گروه‌ها و دسته‌های مجاز',
   693→      desc: 'گروه‌های فعالیت، دسته‌های مجاز و مقادیر گروهی',
   694→    },
   695→    { key: 'ratingTables', label: 'جدول‌های وستینگهاوس', desc: 'جدول‌های مرجع ارزیابی و نمرات آنها' },
   696→    { key: 'rawObservations', label: 'مشاهدات خام', desc: 'داده‌های خام زمان‌سنجی ثبت‌شده' },
   697→  ]
   698→
   699→  const submit = async () => {
   700→    if (!newName.trim()) {
   701→      toast.error('نام پروژه جدید الزامی است')
   702→      return
   703→    }
   704→    setSaving(true)
   705→    try {
   706→      const res = await projectsApi.duplicate(project.id, {
   707→        newProjectName: newName.trim(),
   708→        include: {
   709→          structure: sections.structureAndSettings,
   710→          settings: sections.structureAndSettings,
   711→          operators: sections.operators,
   712→          groupsAndAllowances: sections.groupsAndAllowances,
   713→          ratingTables: sections.ratingTables,
   714→          rawObservations: sections.rawObservations,
   715→        },
   716→      })
   717→      onDone(res.projectId, newName.trim())
   718→    } catch (err) {
   719→      toast.error(err instanceof Error ? err.message : 'کپی پروژه ناموفق بود')
   720→    } finally {
   721→      setSaving(false)
   722→    }
   723→  }
   724→
   725→  return (
   726→    <Dialog
   727→      open
   728→      onOpenChange={(open) => {
   729→        if (!open) onClose()
   730→      }}
   731→    >
   732→      <DialogContent dir="rtl" className="sm:max-w-md">
   733→        <DialogHeader>
   734→          <DialogTitle>کپی پروژه</DialogTitle>
   735→          <DialogDescription>
   736→            از «{project.name}» یک پروژه جدید با بخش‌های انتخابی شما ساخته می‌شود.
   737→          </DialogDescription>
   738→        </DialogHeader>
   739→        <div className="space-y-4">
   740→          <div className="space-y-1.5">
   741→            <Label htmlFor="pi-dup-name">نام پروژه جدید</Label>
   742→            <Input
   743→              id="pi-dup-name"
   744→              value={newName}
   745→              onChange={(e) => setNewName(e.target.value)}
   746→              placeholder="مثلاً: مطالعه خط مونتاژ پمپ (کپی)"
   747→            />
   748→          </div>
   749→          <fieldset className="space-y-1.5">
   750→            <legend className="text-sm font-bold">بخش‌های قابل کپی</legend>
   751→            {rows.map((row) => (
   752→              <label
   753→                key={row.key}
   754→                htmlFor={`pi-dup-${row.key}`}
   755→                className={cn(
   756→                  'flex cursor-pointer items-start gap-2.5 rounded-xl border p-2.5 transition-colors hover:bg-muted/50',
   757→                  sections[row.key] && 'border-brand bg-brand-soft/40'
   758→                )}
   759→              >
   760→                <Checkbox
   761→                  id={`pi-dup-${row.key}`}
   762→                  checked={sections[row.key]}
   763→                  onCheckedChange={(checked) =>
   764→                    setSections((s) => ({ ...s, [row.key]: checked === true }))
   765→                  }
   766→                  className="mt-0.5"
   767→                />
   768→                <span>
   769→                  <span className="block text-sm font-bold">{row.label}</span>
   770→                  <span className="block text-xs text-muted-foreground">{row.desc}</span>
   771→                </span>
   772→              </label>
   773→            ))}
   774→          </fieldset>
   775→          <p className="rounded-xl bg-amber-soft p-3 text-xs leading-6 text-muted-foreground">
   776→            <b className="text-amber">توجه: </b>
   777→            کپی عادی مشاهدات، نتایج محاسبه‌شده و تاریخچه را منتقل نمی‌کند مگر صریحاً انتخاب شوند.
   778→          </p>
   779→        </div>
   780→        <DialogFooter className="flex-row justify-between gap-2">
   781→          <Button variant="outline" onClick={onClose} disabled={saving}>
   782→            انصراف
   783→          </Button>
   784→          <Button onClick={() => void submit()} disabled={saving} className="btn-brand gap-1.5 border-0">
   785→            {saving ? (
   786→              <Loader2 className="size-4 animate-spin" aria-hidden />
   787→            ) : (
   788→              <Copy className="size-4" aria-hidden />
   789→            )}
   790→            {saving ? 'در حال کپی…' : 'ساخت کپی'}
   791→          </Button>
   792→        </DialogFooter>
   793→      </DialogContent>
   794→    </Dialog>
   795→  )
   796→}
