   106→/** حجم پشتیبان با واحدهای فارسی */
   107→function formatBackupSize(bytes: number, prefs: DisplayPrefs): string {
   108→  if (!Number.isFinite(bytes) || bytes < 0) return '—'
   109→  if (bytes >= 1024 * 1024) return `${formatTimeValue(bytes / (1024 * 1024), prefs, 1)} مگابایت`
   110→  if (bytes >= 1024) return `${formatTimeValue(bytes / 1024, prefs, 1)} کیلوبایت`
   111→  return `${formatInt(bytes, prefs)} بایت`
   112→}
   113→
   114→/** تجزیه امن JSON شیء‌شکل */
   115→function parseJsonObject(text: string | null): Record<string, unknown> | null {
   116→  if (!text) return null
   117→  try {
   118→    const parsed: unknown = JSON.parse(text)
   119→    return typeof parsed === 'object' && parsed !== null ? (parsed as Record<string, unknown>) : null
   120→  } catch {
   121→    // JSON نامعتبر — نادیده گرفته می‌شود
   122→    return null
   123→  }
   124→}
   125→
   126→/** نمایش مقدار Audit به‌صورت متن کوتاه */
   127→function fmtAuditValue(value: unknown): string {
   128→  if (value === undefined || value === null) return '—'
   129→  if (value === '') return '(خالی)'
   130→  if (typeof value === 'string') return value
   131→  if (typeof value === 'number' || typeof value === 'boolean') return String(value)
   132→  try {
   133→    return JSON.stringify(value)
   134→  } catch {
   135→    return String(value)
   136→  }
   137→}

  1023→/** نمایش تغییرات قبل/بعد یک رویداد — FR-27 */
  1024→function AuditDiff({ item }: { item: AuditDto }) {
  1025→  if (item.action === 'DELETE') {
  1026→    return (
  1027→      <p className="rounded-xl bg-red-soft/60 p-3 text-xs leading-6 text-muted-foreground">
  1028→        <b className="text-red">حذف دائمی: </b>
  1029→        بدون محتوای حذف‌شده (Audit حداقلی) — فقط نوع موجودیت، شناسه، زمان و عامل ثبت شده است.
  1030→      </p>
  1031→    )
  1032→  }
  1033→  const before = parseJsonObject(item.beforeJson)
  1034→  const after = parseJsonObject(item.afterJson)
  1035→  const keys = [...new Set([...Object.keys(before ?? {}), ...Object.keys(after ?? {})])]
  1036→  if (keys.length === 0) {
  1037→    return (
  1038→      <p className="text-xs leading-6 text-muted-foreground">
  1039→        جزئیات قبل/بعد برای این رویداد ثبت نشده است.
  1040→      </p>
  1041→    )
  1042→  }
  1043→  return (
  1044→    <div className="space-y-1.5">
  1045→      {keys.map((key) => (
  1046→        <div
  1047→          key={key}
  1048→          className="flex flex-wrap items-baseline gap-x-2 gap-y-0.5 rounded-lg bg-muted px-3 py-1.5 text-xs"
  1049→        >
  1050→          <span className="min-w-24 font-bold">{key}:</span>
  1051→          <span className="text-red" title="مقدار قبل">
  1052→            {fmtAuditValue(before?.[key])}
  1053→          </span>
  1054→          <span aria-hidden>←</span>
  1055→          <span className="font-bold text-green" title="مقدار بعد">
  1056→            {fmtAuditValue(after?.[key])}
  1057→          </span>
  1058→        </div>
  1059→      ))}
  1060→    </div>
  1061→  )
  1062→}
