   930→// ═══════════════════════════════════════════════════════════
   931→// ویرایشگر مقادیر مجاز گروه — debounce با useAutosave
   932→// کل values (همه دسته‌ها) در هر ذخیره ارسال می‌شود — FR-04
   933→// ═══════════════════════════════════════════════════════════
   934→function AllowancesEditor({
   935→  groupId,
   936→  groupName,
   937→  allowances,
   938→  projectId,
   939→  prefs,
   940→  isMobile,
   941→}: {
   942→  groupId: string
   943→  groupName: string
   944→  allowances: GroupAllowancesDto
   945→  projectId: string
   946→  prefs: Prefs
   947→  isMobile: boolean
   948→}) {
   949→  const qc = useQueryClient()
   950→
   951→  // درصد کاربر ↔ مقدار اعشاری API — FR-15
   952→  const [values, setValues] = useState<Record<string, number | null>>(() =>
   953→    Object.fromEntries(allowances.values.map((v) => [v.categoryId, v.percentValue]))
   954→  )
   955→
   956→  // همگام‌سازی محافظه‌کارانه با داده سرور (ویرایش‌های کاربر حفظ می‌شود)
   957→  useEffect(() => {
   958→    setValues((prev) => {
   959→      const next: Record<string, number | null> = {}
   960→      for (const v of allowances.values) {
   961→        next[v.categoryId] = v.categoryId in prev ? prev[v.categoryId] : v.percentValue
   962→      }
   963→      return next
   964→    })
   965→  }, [allowances])
   966→
   967→  const { status, retry } = useAutosave<Record<string, number | null>>({
   968→    data: values,
   969→    ready: true,
   970→    delayMs: 900,
   971→    save: async (v) => {
   972→      const entries = Object.entries(v).map(([categoryId, percentValue]) => ({ categoryId, percentValue }))
   973→      if (entries.length === 0) return
   974→      const res = await allowancesApi.putGroupAllowances(groupId, entries)
   975→      qc.setQueryData(qk.groupAllowances(groupId), res)
   976→      void qc.invalidateQueries({ queryKey: qk.groups(projectId) })
   977→      void qc.invalidateQueries({ queryKey: qk.projectSummary(projectId) })
   978→    },
   979→  })
   980→
   981→  const setPercent = (categoryId: string, v: number | null) => {
   982→    setValues((s) => ({ ...s, [categoryId]: v === null ? null : Math.min(v, MAX_PERCENT_DECIMAL) }))
   983→  }
   984→
   985→  return (
   986→    <div className="space-y-3">
   987→      <div className="flex flex-wrap items-center justify-between gap-2">
   988→        <h3 className="flex items-center gap-1 text-sm font-bold">
   989→          مجازهای گروه «{groupName}»
   990→          <HelpDot
   991→            label="مجازهای گروه"

   994→              input: 'عدد درصد وارد کنید (۰ تا ۹۹٫۹۹)؛ خالی بگذارید تا آن دسته مقداری نداشته باشد.',
   995→              example: 'استراحت شخصی ۵٪ یا نیازهای طبیعی ۳٪ — درصد صرفاً مثال است و نرخ مصوب شرکت نیست.',
   996→              note: 'فقط دسته‌های فعال در مجموع A لحاظ می‌شوند؛ درصدها به‌صورت مقدار اعشاری (۰٫۱۵ برای ۱۵٪) ذخیره می‌شوند.',
   997→            }}
   998→          />
   999→        </h3>
  1000→        <SaveIndicator status={status} onRetry={retry} />
  1001→      </div>
  1002→
  1003→      {isMobile ? (
  1004→        /* کارت موبایل */
  1005→        <div className="space-y-3">
  1006→          {allowances.values.map((cat) => (
  1007→            <div key={cat.categoryId} className={cn('card-premium p-4', !cat.isActive && 'opacity-60')}>
  1008→              <div className="flex items-start justify-between gap-2">
  1009→                <div className="min-w-0">
  1010→                  <p className="truncate font-bold">{cat.name}</p>
  1011→                  {cat.code && (
  1012→                    <p className="num text-xs text-faint" dir="ltr">
  1013→                      {cat.code}
  1014→                    </p>
  1015→                  )}
  1016→                </div>
  1017→                {cat.isActive ? (
  1018→                  <Badge className="bg-green-soft text-green">فعال</Badge>
  1019→                ) : (
  1020→                  <Badge variant="outline" className="text-faint">
  1021→                    غیرفعال — در A لحاظ نمی‌شود
  1022→                  </Badge>
  1023→                )}
  1024→              </div>
  1025→              <div className="mt-3 flex max-w-40 items-center gap-2">
  1026→                <PercentInput
  1027→                  value={values[cat.categoryId] ?? null}
  1028→                  onChange={(v) => setPercent(cat.categoryId, v)}
  1029→                  aria-label={`درصد مجاز دسته ${cat.name}`}
  1030→                />
  1031→              </div>
  1032→            </div>
  1033→          ))}
  1034→          <div className="card-premium flex items-center justify-between p-4">
  1035→            <span className="text-sm font-bold">جمع A (دسته‌های فعال)</span>

  1081→                  جمع A (دسته‌های فعال)
