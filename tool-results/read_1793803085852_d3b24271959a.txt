   109→/** نمایش مقدار درجه با علامت ± */
   110→function formatSigned(v: number | null | undefined, prefs: Prefs): string {
   111→  if (v === null || v === undefined || !Number.isFinite(v)) return '—'
   112→  if (Math.abs(v) < 1e-9) return formatTimeValue(0, prefs, 2)
   113→  return (v > 0 ? '+' : '−') + formatTimeValue(Math.abs(v), prefs, 2)
   114→}

   612→// دیالوگ ارزیابی عنصر — FR-13
   613→// جدول + چهار عامل + پیش‌نمایش زنده R + کپی از عنصر مشابه
   614→// ═══════════════════════════════════════════════════════════
   615→interface RatingFormState {
   616→  tableId: string
   617→  grades: Record<RatingFactor, string | null>
   618→  notes: string
   619→  copyFrom: string | null
   620→}
   621→
   622→const EMPTY_GRADES: Record<RatingFactor, string | null> = {
   623→  SKILL: null,
   624→  EFFORT: null,
   625→  CONDITIONS: null,
   626→  CONSISTENCY: null,
   627→}
   628→
   629→function ElementRatingDialog({
   630→  element,
   631→  tables,
   632→  defaultTableId,
   633→  candidateSources,
   634→  projectId,
   635→  prefs,
   636→  onClose,
   637→}: {
   638→  element: MatrixElementRow | null
   639→  tables: RatingMatrixDto['tables']
   640→  defaultTableId: string | null
   641→  candidateSources: { id: string; label: string }[]
   642→  projectId: string
   643→  prefs: Prefs
   644→  onClose: () => void
   645→}) {
   646→  const qc = useQueryClient()
   647→  const open = element !== null
   648→
   649→  // ارزیابی فعلی عنصر
   650→  const ratingQ = useQuery({
   651→    queryKey: qk.elementRating(open ? element.id : 'none'),
   652→    queryFn: () => ratingApi.elementRating(element!.id),
   653→    enabled: open,
   654→  })
   655→
   656→  const [form, setForm] = useState<RatingFormState | null>(null)
   657→  const dirtyRef = useRef(false)
   658→
   659→  const close = () => {
   660→    dirtyRef.current = false
   661→    setForm(null)
   662→    onClose()
   663→  }
   664→
   665→  // مقداردهی اولیه فرم از ارزیابی فعلی (تا زمانی که کاربر ویرایش نکرده است)
   666→  useEffect(() => {
   667→    if (!element || !ratingQ.data || dirtyRef.current) return
   668→    const a = ratingQ.data.assessment
   669→    setForm({
   670→      tableId: a?.ratingTableId ?? defaultTableId ?? tables[0]?.id ?? '',
   671→      grades: {
   672→        SKILL: a?.skillGrade ?? null,
   673→        EFFORT: a?.effortGrade ?? null,
   674→        CONDITIONS: a?.conditionsGrade ?? null,
   675→        CONSISTENCY: a?.consistencyGrade ?? null,
   676→      },
   677→      notes: a?.notes ?? '',
   678→      copyFrom: null,
   679→    })
   680→  }, [element, ratingQ.data, defaultTableId, tables])
   681→
   682→  // درجه‌های جدول انتخابی
   683→  const tableQ = useQuery({
   684→    queryKey: qk.ratingTable(form?.tableId ?? 'none'),
   685→    queryFn: () => ratingApi.table(form!.tableId),
   686→    enabled: open && form !== null && form.tableId !== '',
   687→  })
   688→
   689→  // ارزیابی عنصر مبدأ برای کپی
   690→  const sourceQ = useQuery({
   691→    queryKey: qk.elementRating(form?.copyFrom ?? 'none'),
   692→    queryFn: () => ratingApi.elementRating(form!.copyFrom!),
   693→    enabled: open && form !== null && form.copyFrom !== null,
   694→  })
   695→
   696→  const gradesByFactor = useMemo(() => {
   697→    const m: Record<RatingFactor, RatingGradeDto[]> = { SKILL: [], EFFORT: [], CONDITIONS: [], CONSISTENCY: [] }
   698→    for (const g of tableQ.data?.grades ?? []) {
   699→      if ((RATING_FACTORS as readonly string[]).includes(g.factor)) m[g.factor as RatingFactor].push(g)
   700→    }
   701→    return m
   702→  }, [tableQ.data])
   703→
   704→  // کدهای مؤثر: دستی یا کپی‌شده از عنصر مبدأ
   705→  const effectiveGrades = useMemo((): Record<RatingFactor, string | null> => {
   706→    if (!form?.copyFrom) return form?.grades ?? EMPTY_GRADES
   707→    const a = sourceQ.data?.assessment
   708→    if (!a) return EMPTY_GRADES
   709→    return {
   710→      SKILL: a.skillGrade,
   711→      EFFORT: a.effortGrade,
   712→      CONDITIONS: a.conditionsGrade,
   713→      CONSISTENCY: a.consistencyGrade,
   714→    }
   715→  }, [form, sourceQ.data])
   716→
   717→  // مقدار هر عامل از جدول انتخابی (مطابق منطق سرور)؛ null = نامعلوم/نامعتبر
   718→  const factorValue = (factor: RatingFactor): number | null => {
   719→    if (form?.copyFrom && !sourceQ.data?.assessment) return null // در انتظار بارگذاری عنصر مبدأ
   720→    const code = effectiveGrades[factor]
   721→    if (!code) return 0
   722→    const g = gradesByFactor[factor].find((x) => x.gradeCode === code)
   723→    return g ? g.value : null
   724→  }
   725→
   726→  const values = {
   727→    skill: factorValue('SKILL'),
   728→    effort: factorValue('EFFORT'),
   729→    conditions: factorValue('CONDITIONS'),
   730→    consistency: factorValue('CONSISTENCY'),
   731→  }
   732→  const previewR =
   733→    values.skill === null || values.effort === null || values.conditions === null || values.consistency === null
   734→      ? null
   735→      : computeRatingFactor({
   736→          skill: values.skill,
   737→          effort: values.effort,
   738→          conditions: values.conditions,
   739→          consistency: values.consistency,
   740→        })
   741→
   742→  const saveMutation = useMutation({
   743→    // شناسه عنصر در نتیجه بازگردانده می‌شود تا هنگام فراخوانی callback معتبر بماند
   744→    mutationFn: async () => {
   745→      const elId = element!.id
   746→      const f = form!
   747→      const res = await ratingApi.putElementRating(elId, {
   748→        ratingTableId: f.tableId,
   749→        skillGrade: f.grades.SKILL,
   750→        effortGrade: f.grades.EFFORT,
   751→        conditionsGrade: f.grades.CONDITIONS,
   752→        consistencyGrade: f.grades.CONSISTENCY,
   753→        notes: f.notes.trim() || null,
   754→        copyFromElementId: f.copyFrom,
   755→      })
   756→      return { res, elId }
   757→    },
   758→    onSuccess: ({ res, elId }) => {
   759→      void qc.invalidateQueries({ queryKey: qk.ratingMatrix(projectId) })
   760→      void qc.invalidateQueries({ queryKey: qk.elementRating(elId) })
   761→      void qc.invalidateQueries({ queryKey: qk.projectSummary(projectId) })
   762→      toast.success(`ارزیابی ثبت شد — R = ${formatTimeValue(res.assessment?.totalFactor ?? null, prefs)}`)
   763→      close()
   764→    },
   765→    onError: (err) => toast.error(err instanceof Error ? err.message : 'ثبت ارزیابی ناموفق بود'),
   766→  })
   767→
   768→  return (
   769→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   770→      <DialogContent dir="rtl" className="scroll-slim max-h-[88vh] overflow-y-auto sm:max-w-2xl">
   771→        <DialogHeader>
   772→          <DialogTitle className="flex flex-wrap items-center gap-1.5">
   773→            ارزیابی عنصر «{element?.name ?? ''}»
   774→            <HelpDot
   775→              label="سیاست ارزیابی"
   776→              content={{
   777→                what: 'برای هر عنصر کاری یک ارزیابی با چهار عامل مهارت، تلاش، شرایط و یکنواختی ثبت می‌شود.',
   778→                note: 'ارزیابی برای عنصر یک‌بار ثبت و به تمام مشاهدات همان عنصر از همه اپراتورها اعمال می‌شود — تصمیم مالک محصول؛ سیستم ارزیابی جداگانه اپراتوری جعل نمی‌کند.',
   779→              }}
   780→            />
   781→          </DialogTitle>
   782→          <DialogDescription>
   783→            {element ? `${element.stationName} › ${element.activityName} — ` : ''}
   784→            درجه هر عامل را از جدول مرجع انتخاب کنید؛ ضریب R به‌صورت زنده پیش‌نمایش می‌شود.
   785→          </DialogDescription>
   786→        </DialogHeader>
   787→
   788→        {ratingQ.error ? (
   789→          <EmptyState
   790→            className="py-6"
   791→            title="بارگذاری ارزیابی فعلی ناموفق بود"
   792→            description={ratingQ.error instanceof Error ? ratingQ.error.message : 'خطای نامشخص'}
   793→            action={
   794→              <Button
   795→                variant="outline"
   796→                onClick={() =>
   797→                  element && qc.invalidateQueries({ queryKey: qk.elementRating(element.id) })
   798→                }
   799→              >
   800→                تلاش مجدد
   801→              </Button>
   802→            }
   803→          />
   804→        ) : form === null || ratingQ.isLoading ? (
   805→          <div className="space-y-3">
   806→            <Skeleton className="h-10 w-full" />
   807→            <Skeleton className="h-10 w-full" />
   808→            <Skeleton className="h-10 w-full" />
   809→            <Skeleton className="h-24 w-full" />
   810→          </div>
   811→        ) : (
   812→          <div className="space-y-4">
   813→            {/* جدول ارزیابی */}
   814→            <div className="space-y-1.5">
   815→              <div className="flex items-center gap-1">
   816→                <Label>جدول ارزیابی</Label>
   817→                <HelpDot
   818→                  label="جدول ارزیابی"
   819→                  content={{
   820→                    what: 'جدول مرجعی که کد و مقدار درجه‌های چهار عامل از آن خوانده می‌شود؛ هر ارزیابی به نسخه جدول مورد استفاده ارجاع دارد.',
   821→                    note: 'نسخه جدول در سابقه ارزیابی ثبت می‌شود؛ جدول Seed صرفاً نقطه شروع است و باید توسط کارشناس/رویه شرکت تأیید شود.',
   822→                  }}
   823→                />
   824→              </div>
   825→              <Select
   826→                value={form.tableId}
   827→                onValueChange={(v) => {
   828→                  dirtyRef.current = true
   829→                  setForm((s) => s && { ...s, tableId: v })
   830→                }}
   831→              >
   832→                <SelectTrigger className="w-full" aria-label="جدول ارزیابی">
   833→                  <SelectValue placeholder="جدول را انتخاب کنید" />
   834→                </SelectTrigger>
   835→                <SelectContent>
   836→                  {tables.map((t) => (
   837→                    <SelectItem key={t.id} value={t.id}>
   838→                      <span className="num">
   839→                        {t.name} (نسخه {formatInt(t.version, prefs)})
   840→                        {t.isDefault ? ' — پیش‌فرض' : ''}
   841→                      </span>
   842→                    </SelectItem>
   843→                  ))}
   844→                </SelectContent>
   845→              </Select>
   846→              {tables.length === 0 && (
   847→                <p className="text-xs text-amber">جدول ارزیابی‌ای وجود ندارد؛ ابتدا از زبانه «جدول‌های مرجع» جدول بسازید.</p>
   848→              )}
   849→            </div>
   850→
   851→            {/* کپی از عنصر مشابه — FR-13 */}
   852→            <div className="space-y-1.5">
   853→              <div className="flex items-center gap-1">
   854→                <Label>کپی از عنصر مشابه</Label>
   855→                <HelpDot
   856→                  label="کپی از عنصر مشابه"
   857→                  content={{
   858→                    what: 'درجه‌های چهار عامل یک عنصر دیگر را عیناً به این عنصر کپی می‌کند؛ برای عناصر تکراری در ایستگاه‌های مشابه.',
   859→                    input: 'یک عنصر دارای ارزیابی از همین پروژه انتخاب کنید، یا حالت «انتخاب دستی درجه‌ها» را نگه دارید.',
   860→                    note: 'کد درجه‌ها از عنصر مبدأ خوانده و مقدار آن‌ها از جدول انتخابی بالا محاسبه می‌شود؛ اگر جدول کدی را نداشته باشد، ثبت با خطا مواجه می‌شود.',
   861→                  }}
   862→                />
   863→              </div>
   864→              <Select
   865→                value={form.copyFrom ?? NONE}
   866→                onValueChange={(v) => {
   867→                  dirtyRef.current = true
   868→                  setForm((s) => s && { ...s, copyFrom: v === NONE ? null : v })
