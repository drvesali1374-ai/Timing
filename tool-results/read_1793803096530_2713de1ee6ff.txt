   141→/** ساخت توضیح فارسی شمارش داده‌های حذف‌شده برای toast — اعداد واقعی Backend */
   142→function removedCountsText(prefs: ReturnType<typeof useDisplayPrefs>, parts: Array<[string, number]>): string | undefined {
   143→  const segs = parts.filter(([, n]) => n > 0).map(([label, n]) => `${formatInt(n, prefs)} ${label}`)
   144→  return segs.length > 0 ? `${segs.join('، ')} نیز به‌طور دائمی حذف شدند.` : undefined
   145→}

   374→          processType: activityCreateForm.processType,

   379→      setActivityCreating(false)

   389→    const deps: Record<string, number> = {}
   390→    if (activities > 0) deps['فعالیت‌ها'] = activities
   391→    if (elements > 0) deps['عناصر کاری'] = elements
   392→    if (sessions > 0) deps['جلسات زمان‌سنجی'] = sessions
   393→    return deps
   394→  }, [deletingStation])
   395→
   396→  const activityDeleteDeps = useMemo(() => {
   397→    if (!deletingActivity) return {}
   398→    const elements = activityElementCount(deletingActivity)
   399→    const sessions = deletingActivity._count?.sessions ?? 0
   400→    const deps: Record<string, number> = {}
   401→    if (elements > 0) deps['عناصر کاری'] = elements
   402→    if (sessions > 0) deps['جلسات زمان‌سنجی'] = sessions
   403→    return deps
   404→  }, [deletingActivity])
   405→
   406→  const editorStationName = activityEditor
   407→    ? stations.find((s) => s.id === activityEditor.stationId)?.name ?? ''
   408→    : ''

   411→  const treeProps = {
   412→    stations,
   413→    prefs,
   414→    lineName: lineQuery.data?.lineName ?? null,
   415→    productName: lineQuery.data?.productName ?? null,
   416→    lineLoading: lineQuery.isLoading,
   417→    selectedStationId,
   418→    isStationOpen: (id: string) => openMap[id] !== false,
   419→    onToggleStation: toggleStation,
   420→    onSelectStation: selectStation,
   421→    onEditStation: (station: StationDto) => {
   422→      selectStation(station.id)
   423→      setStationDialogOpen(true)
   424→    },
