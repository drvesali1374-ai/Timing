  1059→function TableDetailPanel({

  1063→  prefs,
  1064→  isMobile,
  1065→  onDelete,
  1066→  onCreated,

  1068→  table: RatingTableDto
  1069→  projectId: string
  1070→  nextVersion: number
  1071→  prefs: Prefs

  1073→  onDelete: () => void

  1077→
  1078→  const detailQ = useQuery({
  1079→    queryKey: qk.ratingTable(table.id),
  1080→    queryFn: () => ratingApi.table(table.id),

  1085→    void qc.invalidateQueries({ queryKey: qk.ratingMatrix(projectId) })

  1093→    data: meta,
  1094→    ready: true,
  1095→    save: async (v) => {
  1096→      if (!v.name.trim()) return // نام خالی معتبر نیست؛ تا تکمیل نام ذخیره نمی‌شود

  1100→        notes: v.notes.trim() || null,
  1101→        expectedUpdatedAt: updatedAtRef.current,

  1106→  })

  1109→  const [switchingDefault, setSwitchingDefault] = useState(false)

  1114→      updatedAtRef.current = res.table.updatedAt
  1115→      invalidate()
  1116→      toast.success(
  1117→        next ? `«${table.name}» جدول پیش‌فرض پروژه شد` : `پیش‌فرض بودن «${table.name}» برداشته شد`

  1120→      invalidate()
  1121→      toast.error(err instanceof Error ? err.message : 'تغییر جدول پیش‌فرض ناموفق بود')
  1122→    } finally {
  1123→      setSwitchingDefault(false)
  1124→    }
  1125→  }
  1126→
  1127→  // ── ویرایش محلی درجه‌ها — پیش‌نویس؛ ذخیره فقط از طریق نسخه جدید ──
  1128→  const [draft, setDraft] = useState<RatingGradeDto[] | null>(null)
  1129→  useEffect(() => {
  1130→    const d = detailQ.data
  1131→    if (!d) return
  1132→    setDraft((prev) => {
  1133→      if (prev === null) return [...d.grades]
  1134→      const byId = new Map(prev.map((g) => [g.id, g]))
  1135→      return d.grades.map((g) => byId.get(g.id) ?? g)

  1137→  }, [detailQ.data])
  1138→
  1139→  const serverGrades = detailQ.data?.grades ?? []
  1140→  const dirtyCount =
  1141→    draft?.reduce((n, g) => {
  1142→      const sv = serverGrades.find((x) => x.id === g.id)
  1143→      const same = sv && sv.label === g.label && sv.value === g.value && (sv.hint ?? '') === (g.hint ?? '')
  1144→      return n + (same ? 0 : 1)
  1145→    }, 0) ?? 0
  1146→
  1147→  const updateGrade = (id: string, patch: Partial<Pick<RatingGradeDto, 'label' | 'value' | 'hint'>>) => {
