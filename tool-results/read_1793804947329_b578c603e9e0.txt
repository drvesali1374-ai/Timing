   190→function removedCountsText(prefs: Prefs, parts: Array<[string, number]>): string | undefined {
   191→  const segs = parts.filter(([, n]) => n > 0).map(([label, n]) => `${formatInt(n, prefs)} ${label}`)

   193→}

   195→export default function ActivityView({ projectId }: { projectId: string }) {
   196→  const view = useAppStore((s) => s.view)

   201→// ═══════════════════════════════════════════════════════════
   202→// صفحه فعالیت — همه قلاب‌ها به‌صورت بدون‌قید فراخوانی می‌شوند
   203→// ═══════════════════════════════════════════════════════════
   204→function ActivityScreen({ projectId, activityId }: { projectId: string; activityId: string }) {
   205→  const prefs = useDisplayPrefs()
   206→  const isMobile = useIsMobile()
   207→  const qc = useQueryClient()
   208→  const navigate = useAppStore((s) => s.navigate)
   209→  const [tab, setTab] = useState<TabKey>('info')
   210→  const [confirmDeleteOpen, setConfirmDeleteOpen] = useState(false)
   211→
   212→  // ── داده‌ها ──
   213→  const elementsQ = useQuery({
   214→    queryKey: qk.elements(activityId),
   215→    queryFn: () => structureApi.elements(activityId),
   216→  })
   217→  const stationId = elementsQ.data?.activity.stationId ?? null
   218→  const activityName = elementsQ.data?.activity.name ?? ''
   219→
   220→  // نام ایستگاه والد برای مسیر — از درخت ساختار (FR-03)
   221→  const treeQ = useQuery({
   222→    queryKey: qk.structure(projectId),
   223→    queryFn: () => structureApi.tree(projectId),
   224→  })
   225→
   226→  // مشخصات کامل فعالیت (شامل _count.sessions و updatedAt تازه)
   227→  const actsQ = useQuery({
   228→    queryKey: qk.activities(stationId ?? 'none'),
   229→    queryFn: () => structureApi.activities(stationId!),
   230→    enabled: stationId !== null,
   231→  })
   232→
   233→  const groupsQ = useQuery({
   234→    queryKey: qk.groups(projectId),
   235→    queryFn: () => allowancesApi.groups(projectId),
   236→  })
   237→  const groups = groupsQ.data?.items ?? []
   238→
   239→  const stationName = useMemo(
   240→    () => treeQ.data?.stations.find((s) => s.id === stationId)?.name ?? null,
   241→    [treeQ.data, stationId]
   242→  )
   243→  const treeActivity = useMemo(
   244→    () => treeQ.data?.stations.flatMap((s) => s.activities).find((a) => a.id === activityId) ?? null,
   245→    [treeQ.data, activityId]
   246→  )
   247→  const activity = useMemo(
   248→    () => actsQ.data?.items.find((a) => a.id === activityId) ?? treeActivity,
   249→    [actsQ.data, treeActivity, activityId]
   250→  )
   251→
   252→  const elements = elementsQ.data?.elements ?? []
   253→  const ratedCount = elements.filter((e) => e.rating != null).length
   254→
   255→  // ── بی‌اعتبارسازی کش ──
   256→  const invalidateActivityScope = () => {
   257→    void qc.invalidateQueries({ queryKey: qk.structure(projectId) })
   258→    void qc.invalidateQueries({ queryKey: ['activities'] })
   259→    void qc.invalidateQueries({ queryKey: ['elements'] })
   260→    void qc.invalidateQueries({ queryKey: qk.groups(projectId) })
   261→    void qc.invalidateQueries({ queryKey: qk.activityAllowances(activityId) })
   262→    void qc.invalidateQueries({ queryKey: qk.projectSummary(projectId) })
   263→    void qc.invalidateQueries({ queryKey: qk.projects() })
   264→    void qc.invalidateQueries({ queryKey: qk.ratingMatrix(projectId) })
   265→  }
   266→
   267→  // ── حذف فعالیت — پس از موفقیت به ساختار بازمی‌گردیم ──
   268→  const deleteActivityMutation = useMutation({
   269→    mutationFn: () => structureApi.deleteActivity(activityId),
   270→    onSuccess: (res) => {
   271→      invalidateActivityScope()
   272→      toast.success(`فعالیت «${activityName}» به‌طور دائمی حذف شد`, {
   273→        description: removedCountsText(prefs, [
   274→          ['عنصر کاری', res.removedCounts.elements],
   275→          ['جلسه زمان‌سنجی', res.removedCounts.sessions],
   276→          ['مشاهده', res.removedCounts.observations],
   277→        ]),
   278→      })
   279→      setConfirmDeleteOpen(false)
   280→      navigate({ name: 'project', projectId, stage: 'structure' })
   281→    },
   282→    onError: (err) => toast.error(err instanceof Error ? err.message : 'حذف فعالیت ناموفق بود'),
   283→  })
   284→
   285→  const goBack = () => navigate({ name: 'project', projectId, stage: 'structure' })
   286→
   287→  // ── بارگذاری / خطای کل صفحه ──
   288→  if (elementsQ.isLoading) {
   289→    return (
   290→      <div className="animate-view-in mx-auto max-w-5xl space-y-4">
   291→        <div className="flex items-center gap-2">
   292→          <Skeleton className="size-9 rounded-lg" />
   293→          <Skeleton className="h-4 w-64" />
   294→        </div>
   295→        <Skeleton className="h-8 w-full rounded-xl" />
   296→        <Skeleton className="h-72 w-full rounded-2xl" />
   297→        <Skeleton className="h-48 w-full rounded-2xl" />
   298→      </div>
   299→    )
   300→  }
   301→
   302→  if (elementsQ.isError) {
   303→    return (
   304→      <div className="animate-view-in mx-auto max-w-3xl">
   305→        <EmptyState
   306→          title="بارگذاری فعالیت ناموفق بود"
   307→          description={elementsQ.error instanceof Error ? elementsQ.error.message : 'خطای نامشخص'}
   308→          action={
   309→            <div className="flex flex-wrap justify-center gap-2">
   310→              <Button variant="outline" onClick={goBack}>
   311→                بازگشت به ساختار
   312→              </Button>
   313→              <Button variant="outline" onClick={() => qc.invalidateQueries({ queryKey: qk.elements(activityId) })}>
   314→                تلاش مجدد
   315→              </Button>
   316→            </div>
   317→          }
   318→        />
   319→      </div>
   320→    )
   321→  }
   322→
   323→  return (
   324→    <div className="animate-view-in mx-auto max-w-5xl space-y-4">
   325→      {/* سربرگ: بازگشت + مسیر + نوع فرایندی + وضعیت + اقدام بعدی — FR-03 */}
   326→      <header className="space-y-3">
   327→        <div className="flex flex-wrap items-center gap-2">
   328→          <Button variant="outline" size="icon" className="shrink-0" onClick={goBack} aria-label="بازگشت به ساختار تولید">
   329→            <ArrowLeft className="size-4" aria-hidden />
   330→          </Button>
   331→          <nav aria-label="مسیر فعالیت" className="min-w-0 flex-1 truncate text-xs text-muted-foreground">
   332→            ساختار ←{' '}
   333→            {treeQ.isLoading ? (
   334→              <Skeleton className="inline-block h-3 w-16 align-middle" />
   335→            ) : (
   336→              stationName ?? '—'
   337→            )}{' '}
   338→            ← <span className="font-bold text-foreground">{activityName}</span>
   339→          </nav>
   340→          {activity ? (
   341→            <Badge
   342→              className="shrink-0 border-0 bg-brand-soft text-brand"
   343→              title={`نوع رویداد فرایندی: ${processLabelOf(activity.processType)}`}
   344→            >
   345→              <span className="num font-bold">{processSymbolOf(activity.processType)}</span>
   346→              {processLabelOf(activity.processType)}
   347→            </Badge>
   348→          ) : (
   349→            <Skeleton className="h-6 w-24 shrink-0 rounded-full" />
   350→          )}
   351→        </div>
   352→
   353→        <p className="num text-xs text-muted-foreground">
   354→          وضعیت: {formatInt(ratedCount, prefs)} از {formatInt(elements.length, prefs)} عنصر ارزیابی‌شده
   355→        </p>
   356→
   357→        {/* اقدام بعدی */}
   358→        {elements.length === 0 ? (
   359→          <div className="flex flex-wrap items-center gap-2 rounded-xl bg-amber-soft/70 px-3 py-2 text-xs">
   360→            <Lightbulb className="size-4 shrink-0 text-amber" aria-hidden />
   361→            <span className="font-bold text-amber">اقدام بعدی:</span>
   362→            <Button
   363→              variant="link"
   364→              size="sm"
   365→              className="h-auto gap-1 p-0 text-xs font-bold text-amber underline-offset-4"
   366→              onClick={() => setTab('elements')}
   367→            >
   368→              اولین عنصر کاری را در زبانه «عناصر کاری» اضافه کنید
   369→            </Button>
   370→          </div>
   371→        ) : ratedCount < elements.length ? (
   372→          <div className="flex flex-wrap items-center gap-2 rounded-xl bg-amber-soft/70 px-3 py-2 text-xs">
   373→            <Lightbulb className="size-4 shrink-0 text-amber" aria-hidden />
   374→            <span className="font-bold text-amber">اقدام بعدی:</span>
   375→            <Button
   376→              variant="link"
   377→              size="sm"
   378→              className="h-auto gap-1 p-0 text-xs font-bold text-amber underline-offset-4"

   774→// ── ویرایشگر بازنویسی مجازها — debounce ~600ms برای هر تغییر ──
   775→function AllowancesOverrideEditor({
   776→  activityId,
   777→  projectId,
   778→  data,
   779→  prefs,
   780→  isMobile,
   781→}: {
   782→  activityId: string
   783→  projectId: string
   784→  data: EffectiveAllowancesDto
   785→  prefs: Prefs
   786→  isMobile: boolean
   787→}) {
   788→  const qc = useQueryClient()
   789→
   790→  // درصد کاربر ↔ مقدار اعشاری API — FR-15
   791→  const [drafts, setDrafts] = useState<Record<string, number | null>>(() =>
   792→    Object.fromEntries(data.categories.map((c) => [c.categoryId, c.overrideValue]))
   793→  )
   794→  const serverRef = useRef<Record<string, number | null>>(
   795→    Object.fromEntries(data.categories.map((c) => [c.categoryId, c.overrideValue]))
   796→  )
   797→
   798→  // همگام‌سازی محافظه‌کارانه با داده سرور: دسته‌هایی که کاربر
   799→  // همین الان تغییر نداده از سرور تازه می‌شوند؛ تغییرات در جریان حفظ می‌شوند
   800→  useEffect(() => {
   801→    const nextServer = Object.fromEntries(data.categories.map((c) => [c.categoryId, c.overrideValue]))
   802→    setDrafts((prev) => {
   803→      const next: Record<string, number | null> = {}
   804→      let changed = false
   805→      for (const c of data.categories) {
   806→        const untouched = !(c.categoryId in prev) || prev[c.categoryId] === serverRef.current[c.categoryId]
   807→        next[c.categoryId] = untouched ? c.overrideValue : prev[c.categoryId]
   808→        if (next[c.categoryId] !== prev[c.categoryId]) changed = true
   809→      }
   810→      return changed ? next : prev
   811→    })
   812→    serverRef.current = nextServer
   813→  }, [data])
   814→
   815→  const [saveStatus, setSaveStatus] = useState<SaveStatus>('idle')
   816→  const [saveError, setSaveError] = useState<string | null>(null)
   817→  const pendingRef = useRef<Record<string, number | null>>({})
   818→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   819→  const inflightRef = useRef(false)
   820→
   821→  useEffect(() => {
   822→    return () => {
   823→      if (timerRef.current) clearTimeout(timerRef.current)
   824→    }
   825→  }, [])
   826→
   827→  const sendOverrides = async (overrides: { categoryId: string; percentValue: number | null }[]) => {
   828→    if (overrides.length === 0) return
   829→    if (inflightRef.current) {
   830→      // درخواست دیگر در جریان است — در صف ادغام می‌شود
   831→      for (const o of overrides) pendingRef.current[o.categoryId] = o.percentValue
   832→      return
   833→    }
   834→    inflightRef.current = true
   835→    setSaveStatus('saving')
   836→    setSaveError(null)
   837→    let ok = false
   838→    try {
   839→      const res = await structureApi.putActivityAllowances(activityId, overrides)
   840→      qc.setQueryData(qk.activityAllowances(activityId), res)
   841→      void qc.invalidateQueries({ queryKey: qk.projectSummary(projectId) })
   842→      void qc.invalidateQueries({ queryKey: qk.projects() })
   843→      setSaveStatus('saved')
   844→      ok = true
   845→    } catch (err) {
   846→      setSaveStatus('error')
   847→      setSaveError(err instanceof Error ? err.message : 'خطای نامشخص در ذخیره‌سازی')
   848→      toast.error(err instanceof Error ? err.message : 'ثبت بازنویسی مجاز ناموفق بود')
   849→      // داده ناموفق برای تلاش مجدد صریح در صف می‌ماند — FR-23
   850→      for (const o of overrides) pendingRef.current[o.categoryId] = o.percentValue
   851→      void qc.invalidateQueries({ queryKey: qk.activityAllowances(activityId) })
   852→    } finally {
   853→      inflightRef.current = false
   854→      if (ok) {
   855→        const pending = Object.entries(pendingRef.current)
   856→        pendingRef.current = {}
   857→        if (pending.length > 0) {
   858→          void sendOverrides(pending.map(([categoryId, percentValue]) => ({ categoryId, percentValue })))
   859→        }
   860→      }
   861→    }
   862→  }
   863→

   866→    setDrafts((s) => ({ ...s, [categoryId]: clamped }))
   867→    pendingRef.current[categoryId] = clamped
   868→    if (timerRef.current) clearTimeout(timerRef.current)
   869→    timerRef.current = setTimeout(() => {
   870→      const entries = Object.entries(pendingRef.current)
   871→      pendingRef.current = {}
   872→      void sendOverrides(entries.map(([cid, percentValue]) => ({ categoryId: cid, percentValue })))
   873→    }, OVERRIDE_DEBOUNCE_MS)
   874→  }

   877→  const removeOverride = (categoryId: string) => {
   878→    delete pendingRef.current[categoryId]
   879→    setDrafts((s) => ({ ...s, [categoryId]: null }))
   880→    void sendOverrides([{ categoryId, percentValue: null }])
   881→  }

   899→      )

   902→      return (
   903→        <Badge className="max-w-40 truncate border-0 bg-brand-soft text-brand" title={cat.sourceLabel}>
   904→          {cat.sourceLabel}
   905→        </Badge>
   906→      )
   907→    }
   908→    return (
   909→      <Badge variant="outline" className="text-faint">
   910→        {cat.sourceLabel}
   911→      </Badge>
   912→    )
   913→  }

   926→      </span>
   927→    </div>
   928→  )
   929→

   939→          variant="ghost"
   940→          size="icon"
   941→          className="size-7 shrink-0 text-red hover:text-red"

   951→

   962→      {isMobile ? (

   964→        <div className="space-y-3">
   965→          {data.categories.map((cat) => (

   968→                <div className="min-w-0">
   969→                  <p className="truncate font-bold">{cat.name}</p>

   973→                    </p>
   974→                  )}
   975→                </div>

   977→              </div>

   989→                <div>

  1004→        <div className="card-premium overflow-hidden">
  1005→          <div className="scroll-slim overflow-x-auto">
  1006→            <table className="w-full text-sm">

  1012→                  <th className="px-4 py-2.5 font-bold">بازنویسی اختصاصی</th>
  1013→                  <th className="px-4 py-2.5 font-bold">مقدار مؤثر</th>
  1014→                </tr>
  1015→              </thead>
  1016→              <tbody>
  1017→                {data.categories.map((cat) => (
  1018→                  <tr key={cat.categoryId} className={cn('border-b last:border-0', !cat.isActive && 'opacity-60')}>
  1019→                    <td className="px-4 py-2">
  1020→                      <span className="font-medium">{cat.name}</span>
  1021→                      {cat.code && (
  1022→                        <span className="num ms-2 text-xs text-faint" dir="ltr">
  1023→                          {cat.code}
  1024→                        </span>
  1025→                      )}
  1026→                      {!cat.isActive && (
  1027→                        <span className="block text-[10px] text-faint">غیرفعال — در A لحاظ نمی‌شود</span>
  1028→                      )}
  1029→                    </td>
  1030→                    <td className="px-4 py-2">{sourceBadge(cat)}</td>
  1031→                    <td className="num px-4 py-2 text-muted-foreground">{formatPercent(cat.groupValue, prefs)}</td>
  1032→                    <td className="w-52 px-4 py-2">{overrideInput(cat)}</td>

  1036→                <tr className="border-t-2 bg-surface-2/70">
  1037→                  <td colSpan={5} className="px-4 py-2.5">
  1038→                    {summaryRow}
  1039→                  </td>
  1040→                </tr>
  1041→              </tbody>
  1042→            </table>
  1043→          </div>
  1044→        </div>
  1045→      )}
  1046→
  1047→      {shareWarning && (
  1048→        <p className="rounded-xl bg-red-soft p-3 text-xs font-bold leading-6 text-red" role="alert">
  1049→          مجموع مجازها باید کمتر از ۱۰۰٪ باشد (قید روش دوم) — جمع فعلی: {formatPercent(data.totalA, prefs)}
  1050→        </p>
  1051→      )}
  1052→    </div>
  1053→  )
  1054→}
  1055→

  1057→// زبانه عناصر کاری — SortableList با D&D + دیالوگ‌ها

  1059→function ElementsSection({

  1119→      if (res.removedCounts.hasRating) parts.push(['ارزیابی وستینگهاوس', 1])

  1142→    qc.setQueryData<{ activity: { id: string; name: string }; elements: ElementDto[] }>(
  1143→      qk.elements(activityId),
  1144→      (old) => {
  1145→        if (!old) return old
  1146→        return {
  1147→          ...old,
  1148→          elements: ids.flatMap((id) => {
  1149→            const e = old.elements.find((x) => x.id === id)
  1150→            return e ? [e] : []
  1151→          }),
  1152→        }
  1153→      }

  1155→    reorderMutation.mutate(ids)

  1157→
  1158→  const submitCreate = async () => {

  1179→  if (counts && counts.observations > 0) deleteDeps['مشاهده'] = counts.observations

  1181→
  1182→  return (

  1189→            {formatInt(elements.length, prefs)}
  1190→          </Badge>
  1191→          <HelpDot label="عناصر کاری" content={ELEMENT_TYPE_HELP} />
  1192→        </h3>
  1193→        <Button onClick={openCreate} className="btn-brand gap-1.5 border-0">
  1194→          <Plus className="size-4" aria-hidden />

  1199→      {elements.length === 0 ? (

  1202→          title="هنوز عنصر کاری ثبت نشده است"
  1203→          description="هر چرخه فعالیت از عناصر کاری تشکیل می‌شود؛ برای زمان‌سنجی (فاز ۳) حداقل به یک عنصر نیاز دارید. نوع عنصر (دستی، خودکار ماشین، انتظار، هم‌زمان) را می‌توانید بعداً نیز تغییر دهید."
  1204→          action={
  1205→            <Button onClick={openCreate} className="btn-brand gap-1.5 border-0">
  1206→              <Plus className="size-4" aria-hidden />
  1207→              ثبت اولین عنصر کاری
  1208→            </Button>

  1214→          onReorder={handleReorder}
  1215→          renderItem={(element, index) => (
  1216→            <ElementRow
  1217→              element={element}
  1218→              index={index}
  1219→              prefs={prefs}
  1220→              isMobile={isMobile}
  1221→              onEdit={() => setEditing(element)}
  1222→              onDelete={() => setDeleting(element)}

  1228→      {/* دیالوگ ایجاد عنصر کاری */}
  1229→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
  1230→        <DialogContent dir="rtl" className="sm:max-w-md">
  1231→          <DialogHeader>
  1232→            <DialogTitle>عنصر کاری جدید</DialogTitle>
  1233→            <DialogDescription>
  1234→              عنصر در انتهای فهرست افزوده می‌شود و با کشیدن‌ورها کردن قابل جابه‌جایی است.
  1235→            </DialogDescription>
  1236→          </DialogHeader>
  1237→          <div className="space-y-4">
  1238→            <div className="space-y-1.5">
  1239→              <div className="flex items-center gap-1">

  1245→                value={createForm.name}

  1309→      <ConfirmDeleteDialog
