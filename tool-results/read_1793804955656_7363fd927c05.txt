   149→   149→function removedCountsText(prefs: Prefs, parts: Array<[string, number]>): string | undefined {
   150→   150→  const segs = parts.filter(([, n]) => n > 0).map(([label, n]) => `${formatInt(n, prefs)} ${label}`)
