   148→/** توضیح فارسی شمارش داده‌های حذف‌شده برای toast — اعداد واقعی Backend */
   149→function removedCountsText(prefs: Prefs, parts: Array<[string, number]>): string | undefined {
   150→  const segs = parts.filter(([, n]) => n > 0).map(([label, n]) => `${formatInt(n, prefs)} ${label}`)
   151→  return segs.length > 0 ? `${segs.join('، ')} نیز به‌طور دائمی حذف شدند.` : undefined
   152→}

   409→  const stationDeleteDeps = useMemo(() => {

   417→      (sum, a) => sum + (a._count?.sessions ?? 0),
   418→      0
   419→    )
   420→    const deps: Record<string, number> = {}
   421→    if (activityCount > 0) deps['فعالیت'] = activityCount
   422→    if (elementCount > 0) deps['عنصر کاری'] = elementCount
   423→    if (sessionCount > 0) deps['جلسه زمان‌سنجی'] = sessionCount
